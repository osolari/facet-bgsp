% SAIM Computer Modern Sans typography system - single source of truth.
% This reproduces the font system used by the reference Beamer deck:
% Computer Modern Sans for text, Computer Modern Typewriter for code, and the
% classic Computer Modern / AMS families for mathematics.
%
% The exact reference rendering is a pdfLaTeX system. No uploaded font files
% are required; all fonts ship with a standard TeX distribution.
\RequirePackage{fix-cm}
\RequirePackage[utf8]{inputenc}
\RequirePackage{textcomp}
\RequirePackage[protrusion=true,expansion=true]{microtype}
\RequirePackage{setspace}

% Use classic Computer Modern Sans everywhere ordinary text is requested.
% Mapping both Roman and Sans roles to cmss also converts legacy explicit
% \rmfamily calls in the template without creating a second visual family.
\renewcommand{\rmdefault}{cmss}
\renewcommand{\sfdefault}{cmss}
\renewcommand{\ttdefault}{cmtt}
\renewcommand{\familydefault}{\sfdefault}

% Computer Modern Sans provides a slanted face rather than a separate italic
% face. Route ordinary italic requests to that native slanted design so prose,
% citations, and code comments remain inside the reference font family.
\DeclareFontShape{OT1}{cmss}{m}{it}{<->ssub*cmss/m/sl}{}
\DeclareFontShape{OT1}{cmss}{bx}{it}{<->ssub*cmss/bx/n}{}
\normalfont

% Semantic font roles. All prose and display text share Computer Modern Sans;
% Mono marks technical literals. Mathematics remains the standard Computer
% Modern math system loaded by LaTeX and the AMS packages.
\providecommand{\SAIMBodyFont}{\sffamily}
\providecommand{\SAIMHeadingFont}{\sffamily\bfseries}
\providecommand{\SAIMCoverTitleFont}{\sffamily\bfseries}
\providecommand{\SAIMCoverSubtitleFont}{\sffamily}
\providecommand{\SAIMNavigationFont}{\sffamily\bfseries}
\providecommand{\SAIMMetaFont}{\sffamily}
\providecommand{\SAIMCodeFont}{\ttfamily}

% Compatibility aliases used by the front-matter implementations.
\providecommand{\SAIMFrontFont}{\sffamily}
\providecommand{\SAIMFrontTitleFont}{\SAIMCoverTitleFont}
\providecommand{\SAIMFrontMetaFont}{\SAIMMetaFont}
\providecommand{\SAIMFrontRagged}{\raggedright\hyphenpenalty=10000\exhyphenpenalty=10000}
